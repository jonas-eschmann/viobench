cmake_minimum_required(VERSION 3.20)
project(solution CXX)
# Build with relpose/solution/runtime/native/build.sh <this directory>  (configures with the wasi-sdk toolchain file inside the image)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(RELPOSE_NATIVE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../relpose/solution/runtime/native" CACHE PATH "directory with relpose_abi.h and relpose_reactor.cmake")
add_executable(solution solution.cpp)
target_compile_options(solution PRIVATE -fno-exceptions)
include(${RELPOSE_NATIVE_DIR}/relpose_reactor.cmake)
relpose_solution(solution)
